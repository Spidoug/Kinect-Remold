cmake_minimum_required(VERSION 3.16)
project(KinectOneRemold LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
add_executable(kinect-one-remold KinectOneRemold.cpp)
target_include_directories(kinect-one-remold PRIVATE ../../shared)
target_link_libraries(kinect-one-remold PRIVATE PkgConfig::LIBUSB pthread)
